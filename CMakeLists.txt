cmake_minimum_required(VERSION 3.25)
project(bemandi LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SPICE_SDK_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/third_party/spice-sdk" CACHE PATH "Directory containing spicesdk.h")
set(NDI_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/third_party/ndi-headers" CACHE PATH "NDI SDK headers")
if(NOT EXISTS "${SPICE_SDK_INCLUDE_DIR}/spicesdk.h")
  message(FATAL_ERROR "Set SPICE_SDK_INCLUDE_DIR")
endif()
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(bemandi SHARED src/plugin.cpp src/d3d9_hooks.cpp src/config.cpp src/log.cpp
    src/capture.cpp src/ndi.cpp src/timing.cpp src/audio.cpp src/game_audio.cpp)
target_include_directories(bemandi PRIVATE include "${SPICE_SDK_INCLUDE_DIR}" "${NDI_INCLUDE_DIR}")
target_compile_definitions(bemandi PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(bemandi PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(bemandi PRIVATE minhook d3d9 d3dcompiler bcrypt)
add_executable(capture_test tests/capture_test.cpp)
target_include_directories(capture_test PRIVATE include)
target_compile_definitions(capture_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(capture_test PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(capture_test PRIVATE d3d9 winmm)
add_executable(ndi_receive_test tests/ndi_receive_test.cpp src/ndi.cpp)
target_include_directories(ndi_receive_test PRIVATE include "${NDI_INCLUDE_DIR}")
target_compile_definitions(ndi_receive_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(ndi_receive_test PRIVATE /W4 /WX /permissive- /EHsc)
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE include)
target_compile_definitions(core_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
add_test(NAME core COMMAND core_test)
foreach(test d3d9_abi audio_queue)
  add_executable(${test}_test tests/${test}_test.cpp)
  target_include_directories(${test}_test PRIVATE include)
  target_compile_definitions(${test}_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${test}_test PRIVATE /W4 /WX /permissive- /EHsc)
  add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
add_executable(audio_integration_test tests/audio_integration_test.cpp src/audio.cpp src/ndi.cpp src/timing.cpp)
target_include_directories(audio_integration_test PRIVATE include "${NDI_INCLUDE_DIR}")
target_compile_definitions(audio_integration_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(audio_integration_test PRIVATE /W4 /WX /permissive- /EHsc)
